========== DATABASE MODEL ==========
import datetime
from sqlalchemy import Column, Integer, String, Float, Boolean, DateTime, ForeignKey, Text, JSON
from sqlalchemy.orm import relationship
from app.database.session import Base

class User(Base):
    __tablename__ = "users"

    id = Column(Integer, primary_key=True, index=True)
    email = Column(String(255), unique=True, index=True, nullable=False)
    hashed_password = Column(String(255), nullable=False)
    full_name = Column(String(255), nullable=False)
    role = Column(String(50), default="ASHA_WORKER")  # ASHA_WORKER, PHC_MO, ADMIN
    is_active = Column(Boolean, default=True)
    created_at = Column(DateTime, default=datetime.datetime.utcnow)

    asha_profile = relationship("AshaWorker", back_populates="user", uselist=False)

class AshaWorker(Base):
    __tablename__ = "asha_workers"

    id = Column(Integer, primary_key=True, index=True)
    user_id = Column(Integer, ForeignKey("users.id"), unique=True, nullable=False)
    worker_id = Column(String(50), unique=True, index=True, nullable=False) # e.g. ASHA-BLR-042
    assigned_area = Column(String(100), default="Whitefield Sector 4")
    phc_name = Column(String(100), default="Kadugodi Primary Health Centre")
    village = Column(String(100), default="Belathur")
    phone = Column(String(20), default="+91-9876543210")
    total_visits = Column(Integer, default=0)

    user = relationship("User", back_populates="asha_profile")
    visits = relationship("HealthcareVisit", back_populates="asha_worker")

class Beneficiary(Base):
    __tablename__ = "beneficiaries"

    id = Column(Integer, primary_key=True, index=True)
    beneficiary_id = Column(String(50), unique=True, index=True, nullable=False) # SYN-001, SYN-002
    name = Column(String(100), index=True, nullable=False)
    category = Column(String(50), nullable=False) # PREGNANT_WOMAN, NEWBORN, INFANT, CHILD
    age_or_details = Column(String(100), nullable=True) # e.g. "24 yrs, 7 months pregnant" or "10 days newborn"
    guardian_name = Column(String(100), nullable=True) # Husband or Mother's name
    village = Column(String(100), default="Belathur Ward 3")
    contact_phone = Column(String(20), default="+91-9988776655")
    danger_flag = Column(Boolean, default=False)
    created_at = Column(DateTime, default=datetime.datetime.utcnow)

    visits = relationship("HealthcareVisit", back_populates="beneficiary")
    alerts = relationship("DangerSignAlert", back_populates="beneficiary")
    followups = relationship("FollowUpTask", back_populates="beneficiary")

class HealthcareVisit(Base):
    __tablename__ = "healthcare_visits"

    id = Column(Integer, primary_key=True, index=True)
    beneficiary_id = Column(Integer, ForeignKey("beneficiaries.id"), nullable=False)
    asha_worker_id = Column(Integer, ForeignKey("asha_workers.id"), nullable=True)
    
    visit_type = Column(String(50), nullable=False) # Newborn Visit, ANC Checkup, Immunization, Postnatal
    visit_date = Column(String(20), default=lambda: datetime.datetime.utcnow().strftime("%Y-%m-%d"))
    weight_kg = Column(Float, nullable=True)
    vaccinations_given = Column(JSON, default=dict) # e.g. {"BCG": "Given", "OPV-0": "Given"}
    health_observations = Column(Text, nullable=True)
    next_visit_date = Column(String(20), nullable=True)
    
    danger_signs_detected = Column(JSON, default=list) # List of detected danger signs
    referral_made = Column(Boolean, default=False)
    referral_facility = Column(String(100), default="Kadugodi PHC")
    
    status = Column(String(50), default="PENDING_CONFIRMATION") # PENDING_CONFIRMATION, CONFIRMED, CANCELLED
    confidence_score = Column(Float, default=0.95)
    transcript_snippet = Column(Text, nullable=True)
    created_at = Column(DateTime, default=datetime.datetime.utcnow)

    beneficiary = relationship("Beneficiary", back_populates="visits")
    asha_worker = relationship("AshaWorker", back_populates="visits")
    alerts = relationship("DangerSignAlert", back_populates="visit")
    followup_task = relationship("FollowUpTask", back_populates="visit", uselist=False)

class DangerSignAlert(Base):
    __tablename__ = "danger_sign_alerts"

    id = Column(Integer, primary_key=True, index=True)
    visit_id = Column(Integer, ForeignKey("healthcare_visits.id"), nullable=True)
    beneficiary_id = Column(Integer, ForeignKey("beneficiaries.id"), nullable=False)
    
    danger_sign_name = Column(String(150), nullable=False)
    severity = Column(String(50), default="EMERGENCY") # EMERGENCY, HIGH
    referral_action = Column(String(255), default="Immediate referral to Primary Health Centre (PHC) now.")
    guideline_citation = Column(String(255), default="MoHFW Operational Guidelines & IMNCI Protocol Section 3.2")
    
    acknowledged = Column(Boolean, default=False)
    referred_to_phc = Column(Boolean, default=True)
    created_at = Column(DateTime, default=datetime.datetime.utcnow)

    beneficiary = relationship("Beneficiary", back_populates="alerts")
    visit = relationship("HealthcareVisit", back_populates="alerts")

class FollowUpTask(Base):
    __tablename__ = "follow_up_tasks"

    id = Column(Integer, primary_key=True, index=True)
    beneficiary_id = Column(Integer, ForeignKey("beneficiaries.id"), nullable=False)
    visit_id = Column(Integer, ForeignKey("healthcare_visits.id"), nullable=True)
    
    task_title = Column(String(150), nullable=False) # e.g. "Postnatal 14-day checkup", "PHC Referral Follow-up"
    due_date = Column(String(20), nullable=False) # YYYY-MM-DD
    status = Column(String(50), default="PENDING") # PENDING, COMPLETED, OVERDUE
    priority = Column(String(50), default="NORMAL") # NORMAL, HIGH, EMERGENCY
    notes = Column(Text, nullable=True)
    completed_at = Column(DateTime, nullable=True)
    created_at = Column(DateTime, default=datetime.datetime.utcnow)

    beneficiary = relationship("Beneficiary", back_populates="followups")
    visit = relationship("HealthcareVisit", back_populates="followup_task")

class ConversationSession(Base):
    __tablename__ = "conversation_sessions"

    id = Column(Integer, primary_key=True, index=True)
    session_id = Column(String(100), unique=True, index=True, nullable=False)
    beneficiary_id = Column(String(50), nullable=True)
    raw_transcript = Column(Text, default="")
    structured_data = Column(JSON, default=dict)
    audio_duration_seconds = Column(Float, default=0.0)
    status = Column(String(50), default="ACTIVE") # ACTIVE, COMPLETED, ABORTED
    created_at = Column(DateTime, default=datetime.datetime.utcnow)

class AuditEvent(Base):
    __tablename__ = "audit_events"

    id = Column(Integer, primary_key=True, index=True)
    event_type = Column(String(100), nullable=False) # RECORD_CONFIRMED, DANGER_SIGN_DETECTED, VOICE_EDIT
    description = Column(String(255), nullable=False)
    user_id = Column(Integer, nullable=True)
    metadata_json = Column(JSON, default=dict)
    created_at = Column(DateTime, default=datetime.datetime.utcnow)

import datetime
from typing import List, Optional
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from app.database.session import get_db
from app.models.models import HealthcareVisit, Beneficiary, DangerSignAlert, FollowUpTask, AshaWorker, AuditEvent

router = APIRouter(prefix="/api/visits", tags=["Visits"])

@router.get("", response_model=List[HealthcareVisitResponse])
def get_visits(db: Session = Depends(get_db)):
    visits = db.query(HealthcareVisit).order_by(HealthcareVisit.created_at.desc()).all()
    results = []
    for v in visits:
        b = db.query(Beneficiary).filter(Beneficiary.id == v.beneficiary_id).first()
        results.append(HealthcareVisitResponse(
            id=v.id,
            beneficiary_id=v.beneficiary_id,
            beneficiary_name=b.name if b else "Unknown",
            beneficiary_code=b.beneficiary_id if b else "SYN-000",
            visit_type=v.visit_type,
            visit_date=v.visit_date,
            weight_kg=v.weight_kg,
            vaccinations_given=v.vaccinations_given or {},
            health_observations=v.health_observations,
            next_visit_date=v.next_visit_date,
            danger_signs_detected=v.danger_signs_detected or [],
            referral_made=v.referral_made,
            referral_facility=v.referral_facility,
            status=v.status,
            confidence_score=v.confidence_score,
            transcript_snippet=v.transcript_snippet,
            created_at=v.created_at
        ))
    return results

@router.post("", response_model=HealthcareVisitResponse)
def create_confirmed_visit(payload: HealthcareVisitCreate, db: Session = Depends(get_db)):
    """
    Saves a structured visit AFTER explicit confirmation.
    Automatically generates follow-up tasks, danger sign alerts, and audit events.
    """
    # 1. Resolve Beneficiary
    b = db.query(Beneficiary).filter(
        (Beneficiary.beneficiary_id.ilike(payload.beneficiary_id_str)) |
        (Beneficiary.name.ilike(f"%{payload.beneficiary_id_str}%"))
    ).first()

    if not b:
        # Fallback to first synthetic beneficiary if not matched
        b = db.query(Beneficiary).first()
        if not b:
            raise HTTPException(status_code=400, detail="Beneficiary could not be resolved.")

    # 2. Get first ASHA profile
    asha = db.query(AshaWorker).first()

    # 3. Create Visit Record
    has_dangers = len(payload.danger_signs_detected) > 0
    visit = HealthcareVisit(
        beneficiary_id=b.id,
        asha_worker_id=asha.id if asha else None,
        visit_type=payload.visit_type,
        visit_date=datetime.datetime.utcnow().strftime("%Y-%m-%d"),
        weight_kg=payload.weight_kg,
        vaccinations_given=payload.vaccinations_given,
        health_observations=payload.health_observations,
        next_visit_date=payload.next_visit_date,
        danger_signs_detected=payload.danger_signs_detected,
        referral_made=has_dangers or payload.referral_made,
        referral_facility="Kadugodi PHC Emergency Unit" if has_dangers else "Kadugodi PHC",
        status="CONFIRMED",
        confidence_score=0.98,
        transcript_snippet=payload.transcript_snippet
    )
    db.add(visit)
    db.flush()

    if asha:
        asha.total_visits = (asha.total_visits or 0) + 1

    # 4. If danger signs detected, trigger Emergency Referral & Alerts
    if has_dangers:
        b.danger_flag = True
        for ds in payload.danger_signs_detected:
            alert = DangerSignAlert(
                visit_id=visit.id,
                beneficiary_id=b.id,
                danger_sign_name=ds.get("name", "Unspecified Danger Sign"),
                severity="EMERGENCY",
                referral_action=ds.get("action_instruction", "Immediate referral to Primary Health Centre (PHC) now."),
                guideline_citation=ds.get("guideline", "MoHFW Operational Guidelines Section 4.1"),
                acknowledged=False,
                referred_to_phc=True
            )
            db.add(alert)

            # Create high-priority follow up for referral
            db.add(FollowUpTask(
                beneficiary_id=b.id,
                visit_id=visit.id,
                task_title=f"URGENT: Verify PHC Referral for {ds.get('name', 'Danger Sign')}",
                due_date=datetime.datetime.utcnow().strftime("%Y-%m-%d"),
                status="PENDING",
                priority="EMERGENCY",
                notes=f"Beneficiary referred to Kadugodi PHC. Verify emergency care delivery."
            ))

    # 5. Routine next visit follow-up task
    if payload.next_visit_date:
        db.add(FollowUpTask(
            beneficiary_id=b.id,
            visit_id=visit.id,
            task_title=f"Scheduled {payload.visit_type} Follow-up",
            due_date=payload.next_visit_date,
            status="PENDING",
            priority="NORMAL",
            notes=f"Vaccination & growth monitoring follow-up."
        ))

    # 6. Audit Event
    db.add(AuditEvent(
        event_type="VISIT_CONFIRMED_AND_SAVED",
        description=f"Visit confirmed for {b.name} ({b.beneficiary_id}). Danger signs: {len(payload.danger_signs_detected)}",
        metadata_json={
            "visit_id": visit.id,
            "beneficiary_id": b.beneficiary_id,
            "weight_kg": payload.weight_kg,
            "danger_signs": payload.danger_signs_detected
        }
    ))

    db.commit()

    return HealthcareVisitResponse(
        id=visit.id,
        beneficiary_id=b.id,
        beneficiary_name=b.name,
        beneficiary_code=b.beneficiary_id,
        visit_type=visit.visit_type,
        visit_date=visit.visit_date,
        weight_kg=visit.weight_kg,
        vaccinations_given=visit.vaccinations_given or {},
        health_observations=visit.health_observations,
        next_visit_date=visit.next_visit_date,
        danger_signs_detected=visit.danger_signs_detected or [],
        referral_made=visit.referral_made,
        referral_facility=visit.referral_facility,
        status=visit.status,
        confidence_score=visit.confidence_score,
        transcript_snippet=visit.transcript_snippet,
        created_at=visit.created_at
    )

    let ws: WebSocket;
    try {
      ws = new WebSocket(wsUrl);
      wsRef.current = ws;

      ws.onopen = () => {
        setWsConnected(true);
        ws.send(JSON.stringify({ type: 'init', language }));
      };

      ws.onmessage = (event) => {
        try {
          const data = JSON.parse(event.data);

          if (data.type === 'tts_audio_ready') {
            playTimbreAudio(data.audio_base64);
          } else if (data.type === 'tts_fallback') {
            speakBrowserText(data.text);
          } else if (data.type === 'record_updated') {
            setRecord(data.record);
            setIsProcessing(false);

            if (data.readback_summary) {
              setConversationHistory((prev) => [
                ...prev,
                {
                  sender: 'assistant',
                  text: data.readback_summary,
                  time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
                },
              ]);
              // Trigger read-back TTS
              speakText(data.readback_summary);
            }
          } else if (data.type === 'danger_sign_detected') {
            const dangers: DetectedDangerSign[] = data.dangers;
            if (dangers.length > 0) {
              setActiveDanger(dangers[dangers.length - 1]);
              setConversationHistory((prev) => [
                ...prev,
                {
                  sender: 'system',
                  text: `EMERGENCY ALERT: ${dangers[dangers.length - 1].name} detected! Referral to PHC required.`,
                  time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
                },
              ]);
              speakText(data.spoken_alert || 'Emergency danger sign detected. Refer to PHC immediately.');
            }
          } else if (data.type === 'record_saved_successfully') {
            setRecord((prev) => ({ ...prev, status: 'Confirmed' }));
            setSaveSuccessMessage(data.message || 'Record confirmed and saved to database!');
            setIsProcessing(false);
            setConversationHistory((prev) => [
              ...prev,
              {
                sender: 'assistant',
                text: 'Record has been successfully saved to the database. Follow-up task added to calendar.',
                time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
              },
            ]);
            speakText('Record confirmed and saved successfully.');
          }
        } catch (e) {
          console.error('Error handling WS message:', e);
        }
      };
